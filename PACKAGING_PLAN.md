    RT.Start = c(10.0, 20.0, 30.0),
    Precursor.Mz = c(500.0, 600.0, 700.0),

  # Test function
  result <- create_validated_dataset(
    proteome_file = test_file,
    apply_quality_filters = FALSE
  )

  unlink(test_file)
